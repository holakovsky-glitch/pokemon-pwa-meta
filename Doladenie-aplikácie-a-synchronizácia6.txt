<script>
const overlay = document.getElementById('overlay');
const startBtn = document.getElementById('startBtn');
const statusEl = document.getElementById('status');
const canvas = document.getElementById('fireworks');
const ctx = canvas.getContext('2d');

let maxTimeoutId = null;
let voiceTimeoutId = null;
let stopFireworks = null;
let isClosing = false;
let audioCtx = null;

function resizeCanvas() {
    canvas.width = window.innerWidth * window.devicePixelRatio;
    canvas.height = window.innerHeight * window.devicePixelRatio;
    ctx.scale(window.devicePixelRatio, window.devicePixelRatio);
}
resizeCanvas();
window.addEventListener('resize', resizeCanvas);

// FANFARE - hlasnejšia
function playFanfare() {
    try {
        audioCtx = new (window.AudioContext || window.webkitAudioContext)();
        const masterGain = audioCtx.createGain();
        masterGain.gain.value = 0.95; // LOUDER master
        masterGain.connect(audioCtx.destination);

        const now = audioCtx.currentTime;
        
        // Triumfálna fanfára
        const melody = [
            {f: 523.25, t: 0, d: 0.25}, // C5
            {f: 659.25, t: 0.25, d: 0.25}, // E5
            {f: 783.99, t: 0.5, d: 0.25}, // G5
            {f: 1046.50, t: 0.75, d: 0.6}, // C6
            {f: 783.99, t: 1.5, d: 0.2},
            {f: 1046.50, t: 1.75, d: 0.8},
        ];

        melody.forEach(note => {
            const osc = audioCtx.createOscillator();
            const gain = audioCtx.createGain();
            osc.type = 'triangle';
            osc.frequency.setValueAtTime(note.f, now + note.t);
            
            gain.gain.setValueAtTime(0, now + note.t);
            gain.gain.linearRampToValueAtTime(0.35, now + note.t + 0.02); // LOUDER note gain
            gain.gain.exponentialRampToValueAtTime(0.001, now + note.t + note.d);
            
            osc.connect(gain);
            gain.connect(masterGain);
            osc.start(now + note.t);
            osc.stop(now + note.t + note.d + 0.1);
        });

        // Bass
        [130.81, 164.81, 196, 261.63].forEach((f, i) => {
            const osc = audioCtx.createOscillator();
            const gain = audioCtx.createGain();
            osc.type = 'sine';
            osc.frequency.value = f;
            gain.gain.setValueAtTime(0, now + i * 0.25);
            gain.gain.linearRampToValueAtTime(0.25, now + i * 0.25 + 0.05);
            gain.gain.exponentialRampToValueAtTime(0.001, now + i * 0.25 + 0.4);
            osc.connect(gain); gain.connect(masterGain);
            osc.start(now + i * 0.25); osc.stop(now + i * 0.25 + 0.5);
        });

        statusEl.textContent = 'Prehráva sa fanfára...';
    } catch(e) {
        console.warn('Audio error:', e);
    }
}

// FIREWORKS - 8 sekúnd
function startFireworks() {
    let particles = [];
    let running = true;
    const endTime = Date.now() + 8000; // 8s namiesto 6s
    const width = window.innerWidth;
    const height = window.innerHeight;

    ctx.clearRect(0, 0, canvas.width, canvas.height);

    function createBurst(x, y, color) {
        const count = 80 + Math.random() * 40;
        for (let i = 0; i < count; i++) {
            const angle = (Math.PI * 2 * i) / count + Math.random() * 0.2;
            const speed = 2 + Math.random() * 5;
            particles.push({
                x, y,
                vx: Math.cos(angle) * speed,
                vy: Math.sin(angle) * speed,
                life: 1,
                decay: 0.012 + Math.random() * 0.008,
                color,
                size: 2 + Math.random() * 2
            });
        }
    }

    function loop() {
        if (!running) return;
        
        ctx.fillStyle = 'rgba(5, 8, 16, 0.25)';
        ctx.fillRect(0, 0, width, height);

        // nové výbuchy počas 8s
        if (Date.now() < endTime && Math.random() < 0.12) {
            const x = Math.random() * width * 0.8 + width * 0.1;
            const y = Math.random() * height * 0.4 + height * 0.1;
            const hue = Math.floor(Math.random() * 360);
            createBurst(x, y, `hsl(${hue}, 100%, 65%)`);
        }

        particles = particles.filter(p => p.life > 0);
        particles.forEach(p => {
            p.x += p.vx;
            p.y += p.vy;
            p.vy += 0.04;
            p.vx *= 0.99;
            p.life -= p.decay;

            ctx.globalAlpha = Math.max(0, p.life);
            ctx.fillStyle = p.color;
            ctx.beginPath();
            ctx.arc(p.x, p.y, p.size * p.life, 0, Math.PI * 2);
            ctx.fill();
        });
        ctx.globalAlpha = 1;

        if (Date.now() < endTime || particles.length > 0) {
            requestAnimationFrame(loop);
        } else {
            running = false;
        }
    }
    loop();

    return () => { running = false; particles = []; ctx.clearRect(0, 0, width, height); };
}

function closeOverlay() {
    if (isClosing) return;
    isClosing = true;
    
    if (stopFireworks) stopFireworks();
    try { speechSynthesis.cancel(); } catch(e) {}
    try { if (audioCtx) audioCtx.close(); } catch(e) {}
    
    overlay.classList.add('opacity-0');
    statusEl.textContent = 'Prehráva sa fanfára...';
    
    setTimeout(() => {
        overlay.classList.add('hidden');
        isClosing = false;
    }, 700);
}

function startCelebration() {
    // reset
    if (maxTimeoutId) clearTimeout(maxTimeoutId);
    if (voiceTimeoutId) clearTimeout(voiceTimeoutId);
    try { speechSynthesis.cancel(); } catch(e) {}
    if (stopFireworks) stopFireworks();
    
    isClosing = false;
    overlay.classList.remove('hidden');
    requestAnimationFrame(() => overlay.classList.remove('opacity-0'));

    // 1. Fanfára okamžite (0s)
    playFanfare();

    // 2. Ohňostroj na 8s
    stopFireworks = startFireworks();

    // 3. MAX timeout 12s
    maxTimeoutId = setTimeout(() => {
        closeOverlay();
    }, 12000);

    // 4. Hlas od 3.2s
    voiceTimeoutId = setTimeout(() => {
        if (!('speechSynthesis' in window)) return;
        
        statusEl.textContent = 'Hovorí sa...';
        
        const utterance = new SpeechSynthesisUtterance('Gratulujem! Test oslavy prebehol úspešne a hlas sa prehral celý!');
        utterance.lang = 'sk-SK';
        utterance.rate = 0.92;
        utterance.pitch = 1.05;
        utterance.volume = 1;

        // Fallback: zavrieť 1s po skončení reči
        utterance.onend = () => {
            statusEl.textContent = 'Dokončené!';
            clearTimeout(maxTimeoutId);
            setTimeout(() => closeOverlay(), 1000);
        };
        
        utterance.onerror = () => {
            clearTimeout(maxTimeoutId);
            setTimeout(() => closeOverlay(), 1000);
        };

        speechSynthesis.speak(utterance);
    }, 3200);
}

startBtn.addEventListener('click', startCelebration);

// Zavrieť kliknutím
overlay.addEventListener('click', (e) => {
    if (e.target === overlay || e.target === canvas) closeOverlay();
});
